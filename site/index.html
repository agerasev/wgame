A window in Rust

#[wgame::window(size = (800, 600))]
async fn main(
    mut window: Window<'_>
) -> wgame::Result<()> {
    while let Some(mut frame) =
        window.next_frame().await?
    {
        frame.clear(color::BLACK);
        frame.present();
    }
    Ok(())
}
